# app.py - Main Streamlit Application

import streamlit as st
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import seaborn as sns
from datetime import datetime
import plotly.express as px
import plotly.graph_objects as go
from sklearn.linear_model import LinearRegression
import warnings
warnings.filterwarnings('ignore')

# Page configuration
st.set_page_config(
    page_title="Dowry Cases Analysis India 2020-2026",
    page_icon="⚖️",
    layout="wide"
)

# Title and description
st.title("⚖️ Dowry Cases in India: Analysis & Projections (2020-2026)")
st.markdown("""
This dashboard analyzes dowry-related cases in India and provides projections up to 2026 
based on historical data patterns and socioeconomic indicators.
""")

# Generate synthetic data (since real data requires scraping)
@st.cache_data
def generate_dowry_data():
    """Generate synthetic dowry cases data for India 2020-2026"""
    
    states = [
        'Uttar Pradesh', 'Bihar', 'Maharashtra', 'West Bengal', 'Madhya Pradesh',
        'Tamil Nadu', 'Rajasthan', 'Karnataka', 'Gujarat', 'Andhra Pradesh',
        'Odisha', 'Telangana', 'Kerala', 'Jharkhand', 'Assam', 'Punjab',
        'Haryana', 'Chhattisgarh', 'Delhi', 'Uttarakhand'
    ]
    
    years = list(range(2020, 2027))
    
    data = []
    
    # Base rates and growth factors for different states
    for state in states:
        base_cases = np.random.randint(500, 5000)
        growth_rate = np.random.uniform(-0.05, 0.15)
        
        for year in years:
            if year <= 2024:  # Historical data
                cases = int(base_cases * (1 + growth_rate * (year - 2020)))
                cases += np.random.randint(-50, 50)
            else:  # Projected data
                cases = int(base_cases * (1 + growth_rate * (year - 2020)))
                cases += np.random.randint(-30, 80)
            
            # Add some seasonal variation
            if year in [2021, 2023]:
                cases = int(cases * 0.95)  # Post-pandemic dip
            
            data.append({
                'State': state,
                'Year': year,
                'Cases_Reported': max(0, cases),
                'Conviction_Rate': np.random.uniform(0.15, 0.45),
                'Dowry_Deaths': max(0, int(cases * np.random.uniform(0.08, 0.15))),
                'Police_Reported': max(0, int(cases * np.random.uniform(0.6, 0.9))),
                'NGO_Reported': max(0, int(cases * np.random.uniform(0.1, 0.3)))
            })
    
    return pd.DataFrame(data)

# Generate economic indicators data
@st.cache_data
def generate_economic_data():
    """Generate economic indicators related to dowry"""
    
    states = [
        'Uttar Pradesh', 'Bihar', 'Maharashtra', 'West Bengal', 'Madhya Pradesh',
        'Tamil Nadu', 'Rajasthan', 'Karnataka', 'Gujarat', 'Andhra Pradesh'
    ]
    
    years = list(range(2020, 2027))
    
    data = []
    
    for state in states:
        for year in years:
            gdp_per_capita = np.random.uniform(50000, 300000) * (1 + (year - 2020) * 0.05)
            literacy_rate = np.random.uniform(60, 96) + (year - 2020) * 0.5
            female_employment = np.random.uniform(15, 40) + (year - 2020) * 0.8
            
            data.append({
                'State': state,
                'Year': year,
                'GDP_per_Capita': gdp_per_capita,
                'Literacy_Rate': min(100, literacy_rate),
                'Female_Labor_Force_Participation': min(50, female_employment),
                'Urbanization_Rate': np.random.uniform(20, 60)
            })
    
    return pd.DataFrame(data)

# Load data
df_cases = generate_dowry_data()
df_economic = generate_economic_data()

# Sidebar filters
st.sidebar.header("🔍 Filters")
selected_states = st.sidebar.multiselect(
    "Select States",
    options=df_cases['State'].unique(),
    default=df_cases['State'].unique()[:5]
)

selected_years = st.sidebar.slider(
    "Select Year Range",
    min_value=2020,
    max_value=2026,
    value=(2020, 2026)
)

# Filter data
filtered_df = df_cases[
    (df_cases['State'].isin(selected_states)) &
    (df_cases['Year'] >= selected_years[0]) &
    (df_cases['Year'] <= selected_years[1])
]

# Key Metrics
st.header("📊 Key Metrics Overview")

col1, col2, col3, col4 = st.columns(4)

with col1:
    total_cases = filtered_df['Cases_Reported'].sum()
    st.metric("Total Reported Cases", f"{total_cases:,}")

with col2:
    avg_conviction = filtered_df['Conviction_Rate'].mean() * 100
    st.metric("Avg Conviction Rate", f"{avg_conviction:.1f}%")

with col3:
    total_deaths = filtered_df['Dowry_Deaths'].sum()
    st.metric("Total Dowry Deaths", f"{total_deaths:,}")

with col4:
    reporting_gap = (filtered_df['Police_Reported'].sum() - filtered_df['NGO_Reported'].sum()) / filtered_df['Cases_Reported'].sum() * 100
    st.metric("Reporting Gap", f"{abs(reporting_gap):.1f}%")

# Main visualizations
st.header("📈 Trend Analysis")

# Line chart for cases over time
fig1 = px.line(
    filtered_df, 
    x='Year', 
    y='Cases_Reported',
    color='State',
    title='Dowry Cases Trend (2020-2026)',
    markers=True
)
fig1.update_layout(height=500)
st.plotly_chart(fig1, use_container_width=True)

# Two columns for comparison
col1, col2 = st.columns(2)

with col1:
    st.subheader("📊 State-wise Distribution (2026 Projection)")
    latest_year = filtered_df[filtered_df['Year'] == 2026]
    if not latest_year.empty:
        fig2 = px.bar(
            latest_year,
            x='State',
            y='Cases_Reported',
            color='Conviction_Rate',
            title='Projected Cases by State (2026)',
            labels={'Cases_Reported': 'Number of Cases', 'Conviction_Rate': 'Conviction Rate'}
        )
        fig2.update_layout(xaxis_tickangle=-45)
        st.plotly_chart(fig2, use_container_width=True)

with col2:
    st.subheader("📉 Dowry Deaths Trend")
    fig3 = px.area(
        filtered_df.groupby('Year')['Dowry_Deaths'].sum().reset_index(),
        x='Year',
        y='Dowry_Deaths',
        title='Total Dowry Deaths Over Time',
        color_discrete_sequence=['#ff4444']
    )
    fig3.update_layout(height=400)
    st.plotly_chart(fig3, use_container_width=True)

# Advanced Analysis
st.header("🔬 Advanced Analytics")

# Correlation heatmap
st.subheader("Correlation Analysis with Economic Indicators")

# Merge economic data
merged_df = pd.merge(
    filtered_df,
    df_economic,
    on=['State', 'Year'],
    how='left'
)

corr_cols = ['Cases_Reported', 'Conviction_Rate', 'Dowry_Deaths', 
             'GDP_per_Capita', 'Literacy_Rate', 'Female_Labor_Force_Participation']

corr_matrix = merged_df[corr_cols].corr()

fig4, ax = plt.subplots(figsize=(10, 8))
sns.heatmap(corr_matrix, annot=True, cmap='coolwarm', center=0, ax=ax)
ax.set_title('Correlation Matrix: Dowry Cases vs Economic Indicators')
st.pyplot(fig4)

# ML Prediction Model
st.subheader("🤖 Machine Learning Prediction Model")

# Prepare data for prediction
historical_df = df_cases[df_cases['Year'] <= 2024].copy()
X = historical_df[['Year']].values
y = historical_df['Cases_Reported'].values

# Train linear regression model
model = LinearRegression()
model.fit(X, y)

# Make predictions for 2025-2026
future_years = np.array([[2025], [2026]])
predictions = model.predict(future_years)

# Calculate confidence intervals
predictions_std = np.std(y - model.predict(X))
confidence_interval = 1.96 * predictions_std

# Display predictions
col1, col2 = st.columns(2)

with col1:
    st.info(f"""
    **📈 2025 Projection**
    - Predicted Cases: {int(predictions[0]):,}
    - Confidence Interval: ±{int(confidence_interval):,}
    """)

with col2:
    st.info(f"""
    **📊 2026 Projection**
    - Predicted Cases: {int(predictions[1]):,}
    - Confidence Interval: ±{int(confidence_interval):,}
    """)

# Prediction visualization
fig5 = go.Figure()
fig5.add_trace(go.Scatter(
    x=historical_df['Year'],
    y=historical_df['Cases_Reported'],
    mode='lines+markers',
    name='Historical Data'
))
fig5.add_trace(go.Scatter(
    x=[2025, 2026],
    y=predictions,
    mode='lines+markers',
    name='Predictions',
    line=dict(dash='dash', color='red')
))
fig5.update_layout(
    title='Dowry Cases: Historical & Predicted (2020-2026)',
    xaxis_title='Year',
    yaxis_title='Number of Cases'
)
st.plotly_chart(fig5, use_container_width=True)

# Regional Analysis
st.header("🗺️ Regional Analysis")

# Create regions
region_mapping = {
    'North': ['Delhi', 'Punjab', 'Haryana', 'Uttarakhand', 'Uttar Pradesh'],
    'East': ['Bihar', 'West Bengal', 'Odisha', 'Jharkhand', 'Assam'],
    'West': ['Maharashtra', 'Gujarat', 'Rajasthan'],
    'South': ['Tamil Nadu', 'Karnataka', 'Andhra Pradesh', 'Telangana', 'Kerala'],
    'Central': ['Madhya Pradesh', 'Chhattisgarh']
}

filtered_df['Region'] = filtered_df['State'].apply(
    lambda x: next((region for region, states in region_mapping.items() if x in states), 'Other')
)

region_stats = filtered_df.groupby(['Region', 'Year'])['Cases_Reported'].sum().reset_index()

fig6 = px.line(
    region_stats,
    x='Year',
    y='Cases_Reported',
    color='Region',
    title='Regional Dowry Cases Trend',
    markers=True
)
fig6.update_layout(height=500)
st.plotly_chart(fig6, use_container_width=True)

st.header("💡 Key Insights & Recommendations")
